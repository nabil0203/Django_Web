# ▶️Click to Show PDF

### `App.jsx`

```jsx
import Invoice from './Invoice'
import { useState } from 'react'

function App() {
 
  const [showPDF, setShowPDF] = useState(false)                             // by default showPDF is false

  function openPdf() {
    
    // setShowPDF(true)                 // only 'shows' the PDF but doesn't 'hide' on clicking
    setShowPDF(!showPDF)                // 'Shows' and 'hides' the PDF on click

  }


  return (
    <>
      
      <button onClick={openPdf} className="bg-blue-500 hover:bg-blue-700 text-white font-bold py-2 px-4 rounded">
        {showPDF ? 'Hide PDF' : 'Show PDF'}                                                                                           {/*if showPDF is true -> show 'Hide PDF' button, else show 'Show PDF' button */}
      </button>

      {showPDF ? <Invoice /> : null}                                                                                                {/*if showPDF is true -> render Invoice component, else show 'null' (nothing)*/}

    </>
  )
}

export default App

```

- `const [showPDF, setShowPDF] = useState(false)` creates a state variable `showPDF` with initial value `false`
- `function openPdf()` toggles the `showPDF` state variable
- `{showPDF ? <Invoice /> : null}` conditionally renders the `Invoice` component based on the `showPDF` state variable
