# 2️⃣ Show data in UI
- `useState` is needed because
    - UI re-render will occur after fetching data
- Then loop through the data
- And show the data in UI

-----------------------------------------


## 📍`Users.jsx`
```jsx
import { useState } from "react"


function Users() {

    // 'users' is a state hook; initially the value is empty array
    // but after fetching data, it will be filled with users data
    const [users, setUsers] = useState([])                                                                  // useState -> re-render UI after fetching data


    // fetching users data
    async function fetchUsers() {
        
        const r = await fetch('https://fake-json-api.mock.beeceptor.com/users')                              // Waiting for response from api endpoint; r is for response
        const json = await r.json()                                                                          // Waiting for JSON format; json is for json format
        setUsers(json)                                                                                       // 👉Setting users data in state
        
    }

  return (
    <div>
        <h1>Users Component</h1>

        <button onClick={fetchUsers}>Fetch Users </button>                                                  {/* Button */}  


        {/* showing users data in UI */}
        {users.map(user => (
            <div key={user.id}>                                                                              {/* Showing users data in UI; key is for unique identification of each user */}
                {user.name} - {user.email}
            </div>
        ))}

    </div>
  )
}

export default Users
```
