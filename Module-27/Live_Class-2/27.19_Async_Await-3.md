# 3️⃣ Show data in UI (with Loading state)
- In every refresh of the page
    - API is called again
    - So loading also happens again

-------------------------------------------------------------

- To handle this we use a `loading` state
    - `useState` hook is used for this

-------------------------------

- Initially loading is `false`
- When the button is clicked, `setLoading` -> `true`
    - then it will show loading message
- After fetching data `setLoading` -> `false`
    - then it will hide loading message                                                                         


-----------------------------------------------------------



## 📍`Users.jsx`
```jsx
import { useState } from "react"


function Users() {

    const [users, setUsers] = useState([])

    const [isLoading, setLoading] = useState(false)                                              // loading state to handle loading; initially false


    async function fetchUsers() {
        setLoading(true)                                                                        // when the button is clicked, loading becomes true

        const r = await fetch('https://fake-json-api.mock.beeceptor.com/users')
        const json = await r.json()
        setUsers(json)

        setLoading(false)                                                                       // when the fetching process is done, loading becomes false
    }

  return (
    <div>
        <h1>Users Component</h1>

        <button onClick={fetchUsers}>Fetch Users </button>


        {isLoading && <p>Loading users...</p>}                                                        {/* showing loading message when loading is true */}

        {users.map(user => (
            <div key={user.id}> 
                {user.name} - {user.email}
            </div>
        ))}

    </div>
  )
}

export default Users
```
