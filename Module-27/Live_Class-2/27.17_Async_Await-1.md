## 📍`Users.jsx`
```jsx

function Users() {

    async function fetchUsers() {                                                                            // async function for fetching users

        const r = await fetch('https://fake-json-api.mock.beeceptor.com/users')                              // Waiting for response from this API endpoint; r is for response
        
        const json = await r.json()                                                                          // Waiting for JSON format; json is for json format

        console.log(json);                                                                                   // showing JSON format in console
        
    }

  return (
    <div>
        <h1>Users Component</h1>

        <button onClick={fetchUsers}>Fetch Users In Console</button>                                      {/* Button for fetching users; showing users in console whenever the button is clicked */} 
        
    </div>
  )
}

export default Users
```
