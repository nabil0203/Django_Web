# 🪝useMemo Hook

- Used to "memoize" extensive calculations
    - to use it multiple times with same value
- Prevents unnecessary calculations
- Kind of like caching any high calculation

--------------------------------



## 🔹Syntax
```jsx
const value = useMemo(() => computeExpensiveValue(a, b), [a, b]);
```

----------------------------------------------

## 📌`About.jsx`
```jsx
import { useMemo, useState } from "react";

function About() {

  const [count, setCount] = useState(0);
  const fruits = ["Apple", "Banana", "Mango", "Orange"];



  // -----useMemo------
  const fruitLength = useMemo(() => {                                                       // useMemo stores the calculated value
    
    console.log("Calculating length...");                                                   // It will only run again if fruits changes
    return fruits.length;

  }, [fruits]);



  return (
    <div>
      <h2>Fruits: {fruitLength}</h2>                                                         {/* Showing memoized value */}

      <button onClick={() => setCount(count + 1)}>
        Re-render ({count})
      </button>

    </div>
  )


}

export default About;
```
