# 🚨Conditional Rendering

- We have a condition
- Then we are rendering based on that condition



--------------------

### ▶️`App.jsx`
```jsx
import './App.css'
import Course from './Courses'


function App() {

//   const isLoggedIn = true;
  const isLoggedIn = false;

  if (isLoggedIn){                          // 👈if condition
    return return (

    <h1>Welcome</h1>

    )
  } 

  else return(

    <h1>Please Log in</h1>                  // 👈else condition

  )
}

export default App
```
