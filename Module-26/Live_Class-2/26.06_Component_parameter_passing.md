# Component Parameter Passing
- Pass values in differnet component


----------------------------

### 1️⃣`App.jsx`
```jsx
function App() {


const courses = ['C', 'C++', 'Python']


return(

  <div>

    {courses.map((course) => (
        < Courses name={course} />                  // 'name' is the parameter 
                                                    // that is passing the values of the array to 'Courses' component
      )
    )
    }

  </div>


)

}
```
 



-----------------



### 2️⃣`Courses.jsx`
```jsx
function Courses( {name} ){                     // name is the parameter that is catching the value from 'App.jsx'

    return(
    
        <div>
    
            Name: {name}                            {/* returning the value to the 'App.jsx' */}
    
        </div>  
    
    )

}

export default Courses
```
