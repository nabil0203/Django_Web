# ▶️Use Context Provider
- Created the provider
- Now add the functionalities inside the Provider
- Then pass the values to the children in the return
--------------------------


## `AuthProvider.jsx`
```jsx
import { createContext, useContext } from "react";

const AuthContext = createContext();

export const useAuth = () => {
    //----------------
    //----------------
};


const AuthProvider = ({ children }) => {


    const register = async (userdata) => {                                      // register added
        console.log("Registering Function")
    };


    const login = async (userdata) => {                                         // login added
        console.log("Login Function")
    }


    const logout = async () => {                                                // logout added
        console.log("Logout Function")
    }




    return (
        <AuthContext.Provider value={{register, login, logout}}>                    // pass the values; this means these things can be used anywhere
            {children}
        </AuthContext.Provider>
    );
}

export default AuthProvider;
```
